cmake_minimum_required(VERSION 3.16...3.20)
project(removert)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
include(cmake/RosCompatibility.cmake)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(image_transport REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(jsoncpp QUIET)
find_package(PCL REQUIRED COMPONENTS common io filters registration kdtree octree)
find_package(OpenCV REQUIRED)
find_package(OpenMP REQUIRED)
add_executable(removert_removert src/removert_main.cpp src/utility.cpp src/RosParamServer.cpp src/Session.cpp src/Removerter.cpp)
target_include_directories(removert_removert PRIVATE include ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
target_ros_dependencies(removert_removert rclcpp sensor_msgs std_msgs nav_msgs
  visualization_msgs cv_bridge image_transport pcl_conversions)
target_link_libraries(removert_removert PRIVATE ${PCL_LIBRARIES} ${OpenCV_LIBRARIES}
  OpenMP::OpenMP_CXX)
install(TARGETS removert_removert DESTINATION lib/${PROJECT_NAME})
add_executable(removert_reconstruct_feature_map src/reconstruct_feature_map.cpp)
target_include_directories(removert_reconstruct_feature_map PRIVATE include ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
target_ros_dependencies(removert_reconstruct_feature_map rclcpp sensor_msgs std_msgs nav_msgs
  visualization_msgs cv_bridge image_transport pcl_conversions)
target_link_libraries(removert_reconstruct_feature_map PRIVATE ${PCL_LIBRARIES}
  ${OpenCV_LIBRARIES} OpenMP::OpenMP_CXX)
install(TARGETS removert_reconstruct_feature_map DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
ament_package()
