cmake_minimum_required(VERSION 3.16...3.20)
project(ros_compatibility_tests)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
get_filename_component(workspace "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${workspace}/src/lio_sam/cmake/RosCompatibility.cmake")
enable_testing()
foreach(package ltslam removert)
  add_executable(${package}_parameters parameters.cpp)
  target_compile_definitions(${package}_parameters PRIVATE LT_MAPPER_RUNTIME="${package}/runtime.hpp")
  target_include_directories(${package}_parameters PRIVATE
    "${workspace}/src/${package}/include" "${workspace}/src/lio_sam/include"
    "${workspace}/src/carmen_ros_bridge/include" "$ENV{CARMEN_HOME}/include")
  target_ros_dependencies(${package}_parameters rclcpp rosgraph_msgs tf2_ros tf2_geometry_msgs)
  add_test(NAME ${package}_parameters COMMAND ${package}_parameters)
endforeach()
