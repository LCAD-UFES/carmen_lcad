cmake_minimum_required(VERSION 3.16...3.20)
project(lio_sam)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
include(cmake/RosCompatibility.cmake)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(jsoncpp QUIET)
find_package(PCL REQUIRED COMPONENTS common filters registration kdtree io)
find_package(pcl_conversions REQUIRED)
find_package(OpenCV REQUIRED)
find_package(OpenMP REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/CloudInfo.msg"
  DEPENDENCIES std_msgs sensor_msgs)
if(COMMAND rosidl_get_typesupport_target)
  rosidl_get_typesupport_target(cloud_info_typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
else()
  set(cloud_info_typesupport "${PROJECT_NAME}__rosidl_typesupport_cpp")
endif()
include(cmake/GtsamCompatibility.cmake)
add_executable(lio_sam_imuPreintegration src/imuPreintegration.cpp)
target_include_directories(lio_sam_imuPreintegration PRIVATE include ${PCL_INCLUDE_DIRS})
target_ros_dependencies(lio_sam_imuPreintegration rclcpp sensor_msgs nav_msgs geometry_msgs
  std_msgs tf2 tf2_ros tf2_geometry_msgs)
target_link_libraries(lio_sam_imuPreintegration PRIVATE ${PCL_LIBRARIES} gtsam)
add_executable(lio_sam_imageProjection src/imageProjection.cpp)
target_include_directories(lio_sam_imageProjection PRIVATE include ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
target_ros_dependencies(lio_sam_imageProjection rclcpp sensor_msgs nav_msgs geometry_msgs
  std_msgs tf2 tf2_ros tf2_geometry_msgs pcl_conversions)
target_link_libraries(lio_sam_imageProjection PRIVATE ${PCL_LIBRARIES} ${OpenCV_LIBRARIES}
  "${cloud_info_typesupport}")
add_executable(lio_sam_featureExtraction src/featureExtraction.cpp)
target_include_directories(lio_sam_featureExtraction PRIVATE include ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
target_ros_dependencies(lio_sam_featureExtraction rclcpp sensor_msgs nav_msgs geometry_msgs
  std_msgs tf2 tf2_ros tf2_geometry_msgs pcl_conversions)
target_link_libraries(lio_sam_featureExtraction PRIVATE ${PCL_LIBRARIES} ${OpenCV_LIBRARIES}
  "${cloud_info_typesupport}")
include(cmake/CudaMapSearch.cmake)
add_executable(lio_sam_mapOptmization src/mapOptmization.cpp src/Scancontext.cpp)
target_include_directories(lio_sam_mapOptmization PRIVATE include ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
target_ros_dependencies(lio_sam_mapOptmization rclcpp sensor_msgs nav_msgs geometry_msgs
  std_msgs visualization_msgs tf2 tf2_ros tf2_geometry_msgs pcl_conversions)
target_link_libraries(lio_sam_mapOptmization PRIVATE ${PCL_LIBRARIES} ${OpenCV_LIBRARIES}
  "${cloud_info_typesupport}" gtsam OpenMP::OpenMP_CXX cuda_map_search)
# The generated message header must exist before any consumer is compiled.
foreach(target lio_sam_imageProjection lio_sam_featureExtraction lio_sam_mapOptmization)
  add_dependencies(${target} ${PROJECT_NAME}__rosidl_generator_cpp)
endforeach()
install(TARGETS lio_sam_mapOptmization DESTINATION lib/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
install(TARGETS lio_sam_featureExtraction lio_sam_imageProjection lio_sam_imuPreintegration DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
