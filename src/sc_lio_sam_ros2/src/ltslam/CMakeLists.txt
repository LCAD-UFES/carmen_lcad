cmake_minimum_required(VERSION 3.16...3.20)
project(ltslam)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
include(cmake/RosCompatibility.cmake)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(image_transport REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(jsoncpp QUIET)
find_package(PCL REQUIRED COMPONENTS common io filters registration kdtree octree)
find_package(OpenCV REQUIRED)
find_package(OpenMP REQUIRED)
include(cmake/GtsamCompatibility.cmake)
add_executable(ltslam_ltslam src/main.cpp src/utility.cpp src/RosParamServer.cpp src/Session.cpp src/LTslam.cpp src/Scancontext.cpp)
target_include_directories(ltslam_ltslam PRIVATE include ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
target_ros_dependencies(ltslam_ltslam rclcpp sensor_msgs std_msgs nav_msgs visualization_msgs
  cv_bridge image_transport pcl_conversions)
target_link_libraries(ltslam_ltslam PRIVATE ${PCL_LIBRARIES} ${OpenCV_LIBRARIES}
  OpenMP::OpenMP_CXX gtsam)
install(TARGETS ltslam_ltslam DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
ament_package()
